\begin{algorithm}[H]
%
\caption{Factorielle de $n$}\label{alg:factorielle}
\SetAlgoLined
%\SetAlgorithmName{Algorithme}{}{}
\KwData{Entier naturel non nul $n$}
\KwResult{$n!$}
%
\Begin
{
    $r = 1$\;
    \For{$i\leftarrow 1$ \KwTo $n$}
    {
        $r \leftarrow r*i$\;
    }
\Return{$r$}
}
%
\end{algorithm}
